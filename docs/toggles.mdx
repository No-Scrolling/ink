## Example

Long labels wrap. The switch stays centred beside the text.

<div className="pattern-example">
<div>

```tsx app/index.tsx
import { useState } from "react";
import { Screen, Toggle } from "ink";

export default function Settings() {
  const [enabled, setEnabled] = useState(true);

  return <Screen title="Settings">
    <Toggle
      label="Notifications"
      value={enabled}
      onChange={setEnabled}
    />
  </Screen>;
}
```

</div>
